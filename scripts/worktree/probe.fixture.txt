import { v } from "convex/values";
import { mutation, query } from "./_generated/server";
export const write = mutation({
  args: { value: v.string() }, returns: v.id("worktreeProbes"),
  handler: async (ctx, args) => ctx.db.insert("worktreeProbes", args),
});
export const read = query({
  args: {}, returns: v.array(v.object({ _id: v.id("worktreeProbes"), _creationTime: v.number(), value: v.string() })),
  handler: async (ctx) => ctx.db.query("worktreeProbes").take(100),
});
